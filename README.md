# 깃 허브 사용 첫번째

git init
git add README.md
git add . (모든 파일)

git commit -m "first commit"

두번쨰 올릴 때

git add README.md
git commit -m "seconde commit"

# 여러번 구간

git remote add origin https://github.com/jvvy0901/test.git #한번만

git branch -M main # main 대신 다른 이름으로 사용가능 한번만 사용

git push -u origin main (main은 브랜치 이름) 여러번 사용
